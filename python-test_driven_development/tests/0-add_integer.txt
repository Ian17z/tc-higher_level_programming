import unittest
add_integer = __import__('add_integer').add_integer


class TestAddInteger(unittest.TestCase):

    def test_add_integers(self):
        self.assertEqual(add_integer(2, 3), 5)
        self.assertEqual(add_integer(-1, 1), 0)

    def test_add_float(self):
        self.assertEqual(add_integer(1.5, 2.5), 3)
        self.assertEqual(add_integer(3.9, 2), 5)

    def test_add_default_b(self):
        self.assertEqual(add_integer(2), 100)

    def test_type_error_a(self):
        with self.assertRaises(TypeError):
            add_integer("2", 3)

    def test_type_error_b(self):
        with self.assertRaises(TypeError):
            add_integer(2, "3")

    def test_type_error_b_default(self):
        with self.assertRaises(TypeError):
            add_integer(2, None)


if __name__ == "__main__":
    unittest.main()
