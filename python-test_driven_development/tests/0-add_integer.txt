import pytest
from 0-add_integer.py import add_integer

def test_soma_inteiros_basica():
    assert add_integer(1, 2) == 3

def test_soma_com_valor_padrao():
    assert add_integer(2) == 100

def test_conversao_de_float_para_int():
    assert add_integer(4.7, 5.2) == 9

def test_erro_quando_a_nao_e_numero():
    with pytest.raises(TypeError, match="a must be an integer"):
        add_integer("1", 2)

def test_erro_quando_b_nao_e_numero():
    with pytest.raises(TypeError, match="b must be an integer"):
        add_integer(1, [2])

@pytest.mark.parametrize("a, b, esperado", [
    (0, 0, 0),
    (-1, 1, 0),
    (-5, -5, -10),
    (100.9, 1, 101),  # int(100.9) é 100
])
def test_diversos_cenarios(a, b, esperado):
    assert add_integer(a, b) == esperado
